1. GENDOW (/root/gendou.sh)
==========================================================================
cat << 'ROUTER_EOF' > /root/gendou.sh
#!/bin/bash
sysctl -w net.ipv4.ip_forward=1 >/dev/null

# Interface IP Addressing
ip addr flush dev eth0; ip addr add 10.85.4.109/30 dev eth0; ip link set eth0 up
ip addr flush dev eth1; ip addr add 10.85.4.113/30 dev eth1; ip link set eth1 up
ip addr flush dev eth2; ip addr add 10.85.4.117/30 dev eth2; ip link set eth2 up
ip addr flush dev eth3; ip addr add 10.85.4.121/30 dev eth3; ip link set eth3 up
ip addr flush dev eth4; ip addr add 10.85.4.125/30 dev eth4; ip link set eth4 up
ip addr flush dev eth5; ip addr add 10.85.4.129/30 dev eth5; ip link set eth5 up
ip addr flush dev eth6; ip addr add 10.85.4.133/30 dev eth6; ip link set eth6 up
ip link set eth7 up
/gns3/bin/udhcpc -i eth7 -n -q 2>/dev/null || true
ip addr flush dev eth8; ip addr add 10.85.4.105/30 dev eth8; ip link set eth8 up

# Static Routes ke Cabang Fuyutsuki
ip route replace 10.85.4.56/29 via 10.85.4.110
ip route replace 10.85.3.192/27 via 10.85.4.110
ip route replace 10.85.4.72/29 via 10.85.4.110
ip route replace 10.85.4.140/30 via 10.85.4.110
ip route replace 10.85.4.144/30 via 10.85.4.110

# Static Routes ke Cabang Shigeru (Ringkasan CIDR: Monolith .80/29 + Radar .88/29 = .80/28)
ip route replace 10.85.4.80/28 via 10.85.4.114
ip route replace 10.85.4.148/30 via 10.85.4.114

# Static Routes ke Cabang Misato
ip route replace 10.85.4.64/29 via 10.85.4.118

# Static Routes ke Cabang Kaji
ip route replace 10.85.4.100/30 via 10.85.4.122

# Static Routes ke Cabang Asuka
ip route replace 10.85.4.96/30 via 10.85.4.126
ip route replace 10.85.4.48/29 via 10.85.4.126
ip route replace 10.85.3.128/26 via 10.85.4.126
ip route replace 10.85.4.152/30 via 10.85.4.126
ip route replace 10.85.4.156/30 via 10.85.4.126

# Static Routes ke Cabang Hikari
ip route replace 10.85.2.0/24 via 10.85.4.130
ip route replace 10.85.3.224/27 via 10.85.4.130
ip route replace 10.85.4.160/30 via 10.85.4.130

# Static Routes ke Cabang Yui
