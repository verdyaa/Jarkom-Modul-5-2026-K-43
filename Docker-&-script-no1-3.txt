gendou.sh
--------------------------------------------------------------------------
cat <<'EOF' > /root/gendou.sh
#!/usr/bin/env bash
set -euo pipefail

WAN_IF="${1:-eth0}"
CIDR="10.85.0.0/16"

sysctl -w net.ipv4.ip_forward=1 >/dev/null
grep -q '^net.ipv4.ip_forward=1' /etc/sysctl.conf || echo 'net.ipv4.ip_forward=1' >> /etc/sysctl.conf

iptables -F && iptables -t nat -F && iptables -X
iptables -P INPUT DROP
iptables -P FORWARD DROP
iptables -P OUTPUT ACCEPT

iptables -A INPUT -i lo -j ACCEPT
iptables -A OUTPUT -o lo -j ACCEPT
iptables -A INPUT -i "$WAN_IF" -p udp --sport 67 --dport 68 -j ACCEPT

iptables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
iptables -A FORWARD -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT
iptables -A FORWARD -s "$CIDR" -o "$WAN_IF" -m conntrack --ctstate NEW,ESTABLISHED,RELATED -j ACCEPT
iptables -A FORWARD -s "$CIDR" -d "$CIDR" ! -o "$WAN_IF" -j ACCEPT
iptables -A INPUT -s "$CIDR" -j ACCEPT

iptables -t nat -A POSTROUTING -s "$CIDR" -o "$WAN_IF" -j MASQUERADE
iptables -S
EOF
chmod +x /root/gendou.sh

# Sesuaikan interface sesuai kabel di topologi GNS3 kamu:
# contoh: eth0 = NAT1, eth1 = Switch-Cmd (10.85.4.105/30), eth2 = Link Fuyutsuki (10.85.4.109/30), dst.
dhclient eth0 2>/dev/null || true
ip addr add 10.85.4.105/30 dev eth1 2>/dev/null || true
ip addr add 10.85.4.109/30 dev eth2 2>/dev/null || true
ip addr add 10.85.4.113/30 dev eth3 2>/dev/null || true
ip addr add 10.85.4.117/30 dev eth4 2>/dev/null || true
ip addr add 10.85.4.121/30 dev eth5 2>/dev/null || true
ip addr add 10.85.4.125/30 dev eth6 2>/dev/null || true
ip addr add 10.85.4.129/30 dev eth7 2>/dev/null || true
ip addr add 10.85.4.133/30 dev eth8 2>/dev/null || true
ip addr add 10.85.4.137/30 dev eth9 2>/dev/null || true

# Rute ke cabang di bawah Gendou (ganti next-hop sesuai IP lawan link /30 masing-masing)
ip route replace 10.85.4.56/29 via 10.85.4.110
ip route replace 10.85.4.80/29 via 10.85.4.114
ip route replace 10.85.4.64/29 via 10.85.4.118
ip route replace 10.85.4.100/30 via 10.85.4.122
ip route replace 10.85.4.96/30 via 10.85.4.126
