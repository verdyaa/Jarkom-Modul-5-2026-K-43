# 1. Quarters
cat << 'EOF' > /root/quarters.sh
#!/bin/bash
ip addr flush dev eth0
ip addr add 10.85.0.2/23 dev eth0
ip link set eth0 up
ip route replace default via 10.85.0.1
echo "nameserver 192.168.122.1" > /etc/resolv.conf
echo "nameserver 8.8.8.8" >> /etc/resolv.conf
EOF
chmod +x /root/quarters.sh && /root/quarters.sh
grep -q "/root/quarters.sh" /root/.bashrc 2>/dev/null || echo "[ -f /root/quarters.sh ] && /root/quarters.sh >/dev/null 2>&1" >> /root/.bashrc
